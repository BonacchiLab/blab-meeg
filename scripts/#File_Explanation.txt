# ============================================================        #FALTA ACABAR 
# PIPELINE DE PRÉ-PROCESSAMENTO MEEG (um sujeito, 5 runs)             #NAO SEI PK É QUE ISTO É UM PYTHON FILE MAS ACHO QUE NAO DEVIA SER 
# ============================================================

# Estrutura de diretorias (criada por create_output_folders):
#   {subject}_Preproc/
#   ├── 00_badch_maxwell/
#   ├── 01_prep_pipeline/
#   ├── 02_artifact_annotations/
#   ├── 03_ica/
#   └── 04_epochs_FINAL/
#   {subject}_Docs/          # relatórios HTML, CSVs, JSONs

# ============================================================
# step 00 – Bad channels + Maxwell filtering (MEG)
# ============================================================
# INPUT:
#   - raw fifs: {subject}_EXP1_MEEG/{subject}_MEEG_1_DurR[1-5].fif
#   - cal_file: {subject}/metadata/.../calibration_meg.dat
#   - ct_file:  {subject}/metadata/.../crosstalk_meg.fif
# PROCESSAMENTO:
#   - find_bad_channels_maxwell (noisy + flat)
#   - maxwell_filter (SSS, alinhamento head position)
# OUTPUT (save_outputs=True):
#   - /00_badch_maxwell/{subject}_badch_maxwell_{durN}.fif
#   - /Docs/00_badch_maxwell_report.html
#   - /Docs/00_badch_maxwell_scores.csv
#   - /Docs/00_badch_maxwell_info.json

# ============================================================
# step 01 – PREP pipeline (EEG only)
# ============================================================
# INPUT:
#   - ficheiros gerados no PASSO 00:
#     /00_badch_maxwell/{subject}_badch_maxwell_{durN}.fif
# PROCESSAMENTO:
#   - extrai EEG, aplica PrepPipeline (line noise, rereferencing,
#     detecção de maus canais com RANSAC, interpolação)
#   - reintegra EEG limpo no MEG (MEG inalterado)
# OUTPUT:
#   - /01_prep_pipeline/{subject}_01_prep_pipeline_{durN}.fif
#   - /Docs/01_prep_pipeline_report.html
#   - /Docs/01_prep_pipeline_info.json

# ============================================================
# step 02 – Detecção de artefactos (músculo, piscar, pausas)
# ============================================================
# INPUT:
#   - ficheiros do PASSO 01:
#     /01_prep_pipeline/{subject}_01_prep_pipeline_{durN}.fif
# PROCESSAMENTO:
#   - muscle: annotate_muscle_zscore (110‑140 Hz)
#   - blinks: find_eog_events + anotações fixas
#   - breaks: annotate_break (longos períodos sem eventos)
#   - merge annotations no raw
# OUTPUT:
#   - /02_artifact_annotations/{subject}_02_artifact_annotations_{durN}.fif
#   - /Docs/02_artifact_annotations_times.csv   # timings de cada artefacto
#   - /Docs/02_artifact_annotations_report.html

# ============================================================
# step 03 – ICA (treino + aplicação)
# ============================================================
# 3A – TREINO (run_train_ica)
# ----------------------------
# INPUT:
#   - ficheiros do PASSO 02:
#     /02_artifact_annotations/{subject}_02_artifact_annotations_{durN}.fif
# PROCESSAMENTO:
#   - load, pick MEG+EEG+EOG+ECG, filtragem 1‑40 Hz, resample 250 Hz
#   - concatena todos os runs num único raw (para treino)
#   - ICA (FastICA, 99% variância) separada para MEG e para EEG
#   - deteção automática de componentes EOG/ECG (correlação)
# OUTPUT:
#   - /03_ica/{subject}_03_ica_train_file.fif          # raw concatenado usado no treino
#   - /03_ica/{subject}_ica_meg.fif                    # objeto ICA (MEG)
#   - /03_ica/{subject}_ica_eeg.fif                    # objeto ICA (EEG)
#   - /Docs/{subject}_ica_comps_to_remove.json         # componentes auto‑detetadas
#   - /Docs/03_ica_report.html                         # relatório com scores e componentes
#
# (Após o treino, o utilizador deve inspecionar manualmente as componentes e editar
#  o ficheiro JSON, adicionando componentes a "manual": {"eog": [...], "ecg": [...]} )

# 3B – APLICAÇÃO (run_apply_ica)
# --------------------------------
# INPUT:
#   - ficheiros do PASSO 02 (os mesmos anotados)
#   - objetos ICA treinados: /03_ica/*ica_*.fif
#   - JSON com componentes a remover (após edição manual)
# PROCESSAMENTO:
#   - carrega ICA e JSON, define exclude = auto + manual
#   - aplica ICA a cada run (remove componentes)
#   - concatena os 5 runs limpos
# OUTPUT:
#   - /03_ica/{subject}_03_ica_concat.fif              # dados concatenados após ICA
#   - (não guarda os runs individuais aplicados)
#   - relatório: neste script o save está omitido; pode adicionar-se.

# ============================================================
# step 04 – Epoching e baseline
# ============================================================
# INPUT:
#   - ficheiro concatenado do PASSO 03:
#     /03_ica/{subject}_03_ica_concat.fif
# PROCESSAMENTO:
#   - find_events (STI101), extrai eventos entre 1‑80
#   - cria epochs: tmin=-0.9 s, tmax=1.5 s
#   - rejeita epochs com anotações (artifact annotations)
#   - aplica baseline (-0.9 a 0 s)
#   - drop bad epochs baseado em reject_criteria (valores predefinidos)
#   - gera metadata (categoria, orientação, duração, relevância, sexo)
# OUTPUT:
#   - /04_epochs_FINAL/{subject}_04_epochs_FINAL.fif
#   - /Docs/04_epochs_report.html
#   - /Docs/metadata.csv
